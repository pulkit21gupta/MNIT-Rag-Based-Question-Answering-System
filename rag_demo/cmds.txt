----------------------------------------------------------------
 CHECK WHICH FILES ARE INDEXED IN THE DATABASE
----------------------------------------------------------------

    python -c "
    import chromadb, os
    c = chromadb.PersistentClient(path=r'D:\Rag Model\rag\chroma_db')
    for col in c.list_collections():
        items = col.get(include=['metadatas'])
        files = sorted(set(os.path.basename(m.get('source','')) for m in items['metadatas']))
        print(f'{col.name}: {col.count()} chunks')
        [print(f'  - {f}') for f in files]
    "


----------------------------------------------------------------
 VERIFY GROQ API KEY IS WORKING
----------------------------------------------------------------

    python -c "
    from groq import Groq; from dotenv import load_dotenv; import os
    load_dotenv()
    r = Groq(api_key=os.getenv('GROQ_API_KEY')).chat.completions.create(
        model='openai/gpt-oss-120b',
        messages=[{'role':'user','content':'Say OK'}], max_tokens=5)
    print('API Key OK -', r.choices[0].message.content)
    "
